(() => {
  const t0 = Date.now();
  console.log('--- 开始诊断 ---');
  console.log('1) 页面现在的状态:', document.documentElement.getAttribute('data-auto-theming') === null ? 'CMS 未启动' : 'CMS 已启动');
  console.log('2) 页面上的 script 标签:', [...document.scripts].map(s => s.src || 'inline').join(' | '));
  console.log('3) 已有 #cmsFallback:', !!document.getElementById('cmsFallback'));
  const s = document.createElement('script');
  s.type = 'module';
  s.src = 'https://cdn.jsdelivr.net/npm/@sveltia/cms@0.213.2/dist/sveltia-cms.mjs';
  s.onload = () => console.log('4) 手工注入的脚本 onload 触发，用时 ' + (Date.now() - t0) + 'ms');
  s.onerror = (e) => console.log('4) 手工注入的脚本 onerror！', e && e.message ? e.message : '(无细节)');
  document.body.appendChild(s);
  setTimeout(() => {
    console.log('5) 3 秒后：CMS 启动标记 =', document.documentElement.getAttribute('data-auto-theming'),
                '| body 元素数 =', document.body.querySelectorAll('*').length,
                '| 可见文字 =', JSON.stringify((document.body.innerText || '').slice(0, 60)));
  }, 3000);
})()
